\chapter{Starters}
\section{Test Garlic Mushrooms}
\begin{recipeIngredients}{1}
  \item 500g Mushrooms
  \item 4 cloves Garlic
  \item 50g Butter
  \item Fresh Parsley
  \item Salt and Pepper
  \item Splash of White Wine
\end{recipeIngredients}
\subsection*{Method}
\begin{enumerate}
  \item Clean the mushrooms and slice them thinly.
  \item Melt butter in a pan and add crushed garlic.
  \item Sauté mushrooms until golden brown.
  \item Deglaze with wine and garnish with parsley.
\end{enumerate}
