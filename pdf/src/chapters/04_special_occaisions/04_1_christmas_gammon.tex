\section{Baked Gammon with Pineapple Glaze}
\subsection*{Ingredients}
\begin{itemize}
    \item Gammon
    \item Fresh pineapple - fibrous core removed
    \item Spices - cloves, cinnamon, nutmeg, or mixed spice
    \item Ginger - fresh and/or ground
    \item Honey
    \item Vinegar (mild)
    \item Marmelade
\end{itemize}

\subsection*{Instructions}
\begin{enumerate}
    \item Pressure cook the gammon for 20 minutes per 500g plus 20 minutes.
    \item Blend the other ingredients to make a glaze.
\end{enumerate}

\subsection*{Hints}
\begin{itemize}
    \item Buy the pineapple in advance, to make sure it's ripe.
\end{itemize}
