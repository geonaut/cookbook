\subsection*{CHIMMICHURRI}
\subsection*{Ingredients}
\begin{multicols}{2}
\begin{ingredients}
    \item garlic
    \item parsley
    \item oregano
    \item red wine vinegar
    \item olive oil
    \item salt
    \item pepper
\end{ingredients}
\end{multicols}
\subsection*{Instructions}
\begin{enumerate}
    \item Chop all ingredients finely and mix together. Adjust seasoning to taste. Serve with grilled meats.
\end{enumerate}
