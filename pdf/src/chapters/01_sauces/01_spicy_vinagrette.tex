\section{Spicy vinagrette}
\subsection*{Ingredients}
\begin{ingredients}
    \item red wine vinegar
    \item olive oil
    \item garlic
    \item fresh chilli
    \item green pepper
    \item fresh tomato
    \item onion
    \item salt
    \item pepper
\end{ingredients}

\subsection*{Recipe}
\begin{enumerate}
    \item Combine all ingredients in a bowl and whisk until well mixed. Adjust seasoning to taste.
\end{enumerate}
