\subsection*{POL SAMBOL}
\subsection*{Ingredients}
\begin{multicols}{2}
\begin{ingredients}
    \item shallots
    \item grated fresh coconut
    \item red chilli
    \item lime juice
    \item salt
\end{ingredients}
\end{multicols}
\subsection*{Instructions}
\begin{enumerate}
    \item Put all the ingredients into a pestle and mortar and pound until well combined. Adjust seasoning to taste. Serve as a condiment with rice and curry dishes.
\end{enumerate}
