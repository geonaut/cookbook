\section{Chimmichurri}
\subsection*{Ingredients}
\begin{ingredients}
    \item garlic
    \item parsley
    \item oregano
    \item red wine vinegar
    \item olive oil
    \item salt
    \item pepper
\end{ingredients}

\subsection*{Recipe}
\begin{enumerate}
    \item Chop all ingredients finely and mix together. Adjust seasoning to taste. Serve with grilled meats.
\end{enumerate}
