\chapter{Starters}

\section{Bruschetta}
\subsection*{Ingredients}
\begin{itemize}
    \item 1 French baguette
    \item 4 ripe tomatoes, diced
    \item 2 cloves garlic, minced
    \item 1/4 cup fresh basil, chopped
    \item 2 tablespoons olive oil
    \item Salt and pepper to taste
\end{itemize}

\subsection*{Instructions}
\begin{enumerate}
    \item Preheat the oven to 200°C (400°F).
    \item Slice the baguette into 1/2 inch thick slices and place them on a baking sheet.
    \item Brush the slices with olive oil and toast in the oven for about 5-7 minutes, or until golden brown.
    \item In a bowl, combine the diced tomatoes, minced garlic, chopped basil, olive oil, salt, and pepper.
    \item Spoon the tomato mixture onto the toasted baguette slices.
    \item Serve immediately and enjoy!
\end{enumerate}

\subsection*{Recipe Hints}
\begin{itemize}
    \item For added flavor, drizzle balsamic glaze over the bruschetta before serving.
    \item You can also add mozzarella cheese for a richer taste.
\end{itemize}
